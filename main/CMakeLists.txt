target_compile_options(${COMPONENT_LIB} PRIVATE -O3)

# Network settings come from the project's .env file (loaded by the top-level CMakeLists.txt) and are written to a
# generated header, so credentials never enter the source tree.
function(ethoric_env out name default)
    if(DEFINED ENV{${name}} AND NOT "$ENV{${name}}" STREQUAL "")
        set(v "$ENV{${name}}")
    else()
        set(v "${default}")
    endif()
    set(${out} "${v}" PARENT_SCOPE)
endfunction()

ethoric_env(WIFI_SSID WIFI_SSID "")
ethoric_env(WIFI_PASSWORD WIFI_PASSWORD "")
ethoric_env(AP_SSID AP_SSID "Ethoric-ESP32")
ethoric_env(AP_PASSWORD AP_PASSWORD "ethoric123")
ethoric_env(HTTP_PORT HTTP_PORT "8080")
if(WIFI_SSID STREQUAL "your_wifi_name")
    message(WARNING "WIFI_SSID in .env is still the placeholder: the board will create its own access point instead.")
    set(WIFI_SSID "")
endif()

set(WIFI_HEADER "#pragma once\n// Generated by main/CMakeLists.txt from .env. Do not edit or commit.\n")
foreach(key WIFI_SSID WIFI_PASSWORD AP_SSID AP_PASSWORD)
    string(REPLACE "\\" "\\\\" esc "${${key}}")
    string(REPLACE "\"" "\\\"" esc "${esc}")
    string(APPEND WIFI_HEADER "#define ETHORIC_${key} \"${esc}\"\n")
endforeach()
string(APPEND WIFI_HEADER "#define ETHORIC_HTTP_PORT ${HTTP_PORT}\n")
set(WIFI_HEADER_PATH "${CMAKE_CURRENT_BINARY_DIR}/wifi_config.h")
set(old "")
if(EXISTS "${WIFI_HEADER_PATH}")
    file(READ "${WIFI_HEADER_PATH}" old)
endif()
if(NOT old STREQUAL WIFI_HEADER)   # rewrite only when it changed, to avoid needless rebuilds
    file(WRITE "${WIFI_HEADER_PATH}" "${WIFI_HEADER}")
endif()
